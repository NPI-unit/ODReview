# torch is pulled in transitively by cellpose with sensible defaults:

#
# For NVIDIA GPU acceleration on macOS-x86_64 / Linux, install torch *separately*
# from the CUDA wheel index before running this file:
#
#   pip install torch --index-url https://download.pytorch.org/whl/cu121
#
# (replace cu121 with the CUDA version your driver supports)
